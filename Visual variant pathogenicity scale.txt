# ============================================================
# 5. NORMALIZATION FUNCTION
# ============================================================

normalize_value <- function(
  x,
  min_val,
  max_val,
  higher_pathogenic
) {

  if (
    is.na(x) ||
    is.na(min_val) ||
    is.na(max_val)
  ) {
    return(NA_real_)
  }

  if (max_val == min_val) {
    return(NA_real_)
  }

  if (higher_pathogenic) {

    value <- (x - min_val) / (max_val - min_val)

  } else {

    value <- (max_val - x) / (max_val - min_val)
  }

  value <- max(0, min(1, value))

  return(value)
}

# ============================================================
# 10. CLASSIFICATION FUNCTION
# ============================================================

classify_value <- function(
  score,
  uncertain_threshold,
  pathogenic_threshold
) {

  if (is.na(score)) {
    return("NA")
  }

  if (is.na(pathogenic_threshold)) {
    return("NA")
  }

  if (is.na(uncertain_threshold)) {

    if (score >= pathogenic_threshold) {
      return("Pathogenic")
    } else {
      return("Benign")
    }
  }

  if (score < uncertain_threshold) {
    return("Benign")
  }

  if (score < pathogenic_threshold) {
    return("Uncertain")
  }

  return("Pathogenic")
}

  classify_value,
  score = ref$Normalized_score,
  uncertain_threshold = ref$Uncertain_norm,
  pathogenic_threshold = ref$Pathogenic_norm
)
